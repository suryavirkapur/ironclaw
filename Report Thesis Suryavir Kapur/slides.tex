\section{Introduction}

\begin{frame}{Motivation}
  \begin{columns}[T,totalwidth=\textwidth]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item LLM agents excel at long, multi-step workflows.
        \item Fixed tool registries limit what an agent can do at runtime.
        \item \alert{Dynamic tool synthesis} lets the agent generate task-specific code on demand.
        \item Generated code is untrusted and requires strong isolation.
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \begin{tikzpicture}[node distance=0.7cm]
        \node[box, fill=IronBlue!15, minimum width=4cm] (fixed) {\faLock\ Fixed tools};
        \node[host, below=of fixed, minimum width=4cm] (dynamic) {\faCogs\ Dynamic synthesis};
        \node[warn, below=of dynamic, minimum width=4cm] (risk) {\faExclamationTriangle\ Untrusted code};
        \draw[arrow] (fixed) -- (dynamic);
        \draw[arrow] (dynamic) -- (risk);
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Research Questions}
  \begin{block}{\faQuestionCircle\ RQ1: Utility}
    Do agents with secure dynamic tool synthesis outperform fixed-tool agents on complex tasks?
  \end{block}
  \vspace{0.4cm}
  \begin{block}{\faQuestionCircle\ RQ2: Security and Systems}
    What containment and performance trade-offs arise when using Firecracker microVMs instead of process-level sandboxing?
  \end{block}
\end{frame}

\begin{frame}{Enterprise Reality: Why OpenClaw Is Not Enough}
  \small
  \textit{Perspective from experience at Deriv, Fidelity, and Deloitte consulting the world's top insurance companies.}\\[0.15cm]
  \begin{columns}[T,totalwidth=\textwidth]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\faExclamationTriangle\ OpenClaw gaps}
        \begin{itemize}
          \item Process-level sandbox shares the host kernel.
          \item Generated code can persist state or access host resources.
          \item Difficult to prove containment to auditors.
          \item One compromise can affect the whole agent host.
        \end{itemize}
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{\faLock\ IronClaw fix}
        \begin{itemize}
          \item Firecracker microVMs provide hardware-isolated guests.
          \item Each tool runs in a disposable guest with explicit inputs.
          \item Bounded, auditable execution envelope.
          \item Host secrets and workspace remain outside the guest.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.1cm}
  \centering
  \footnotesize\textbf{Enterprises need agents, but they need them behind a strong isolation boundary.}
\end{frame}

\begin{frame}{What Sandboxing Does}

        \item Makes execution auditable and repeatable.

\begin{frame}{Isolation Options}
  \centering
  \renewcommand{\arraystretch}{1.4}
  \begin{tabular}{>{\raggedright\arraybackslash}p{0.18\textwidth} >{\raggedright\arraybackslash}p{0.30\textwidth} >{\raggedright\arraybackslash}p{0.36\textwidth}}
    \toprule
    \textbf{Technique} & \textbf{Boundary} & \textbf{Risk for generated code} \\
    \midrule
    Process sandbox & Same host kernel & Kernel and permission mistakes matter \\
    Container & Namespaces/cgroups, same kernel & Strong packaging, weaker security boundary \\
    Virtual machine & Guest kernel plus hypervisor & Stronger host/guest separation \\
    MicroVM & Minimal VM with small device model & Strong boundary with lower startup overhead \\
    \bottomrule
  \end{tabular}
\end{frame}

\begin{frame}{Expected Outcome}
  \begin{itemize}
    \item Dynamic tools should improve completion on transformation-heavy tasks.
    \item Firecracker microVMs should contain untrusted generated code better than process-only sandboxes.
    \item The main cost is startup latency and memory overhead.
    \item IronClaw makes the utility--security trade-off measurable.
  \end{itemize}
\end{frame}
